  [[52, 9, 51, 6], [33, 2, 13, 53], [47, 50, 49, 7], [18, 48, 35, 14]],

 PI3D = [
 0x02, 0x03, 0x23, 0x3C, 0x33, 0x29, 0x1D, 0x2E,

const getRotValue = (i) => ROT3D[i >> 4][(i >> 2) & 3][i & 3];

const simulatePiRho = (state) => {
  const tmp = new Array(64).fill(0n);
  for (let i = 0; i < 64; i++) {
    const r = getRotValue(i);
    const val = state[i];
    const rotated = r === 0 ? val : ((val << BigInt(r)) | (val >> BigInt(64 - r))) & 0xFFFFFFFFFFFFFFFFn;
    tmp[PI3D[i]] = rotated;
  }
  return tmp;
};

const PI3D_INV = (() => {
  const inv = new Array(64);
  for (let i = 0; i < 64; i++) inv[PI3D[i]] = i;
  return inv;
})();

const getCycles = () => {
  const visited = new Array(64).fill(false);
  const cycles = [];
  for (let start = 0; start < 64; start++) {
    if (visited[start]) continue;
    const cycle = [];
    let i = start;
    while (!visited[i]) { visited[i] = true; cycle.push(i); i = PI3D[i]; }
    cycles.push(cycle);
  }
  return cycles;
};

const getTransitionMatrixMultiRound = (rounds) => {
  const finalDest = new Array(64);
  for (let i = 0; i < 64; i++) {
    let pos = i;
    for (let r = 0; r < rounds; r++) pos = PI3D[pos];
    finalDest[i] = pos;
  }
  const rowMatrix = Array(8).fill(null).map(() => Array(8).fill(0));
  const colMatrix = Array(8).fill(null).map(() => Array(8).fill(0));
  for (let i = 0; i < 64; i++) {
    rowMatrix[Math.floor(i / 8)][Math.floor(finalDest[i] / 8)]++;
    colMatrix[i % 8][finalDest[i] % 8]++;
  }
  return { rowMatrix, colMatrix };
};

const testDiffusionFrom = (startPos, rounds) => {
  const results = [];
  for (let r = 1; r <= rounds; r++) {
    const s0 = new Array(64).fill(0n); s0[startPos] = 1n;
    const s1 = new Array(64).fill(0n); s1[startPos] = 2n;
    let a = s0, b = s1;
    for (let i = 0; i < r; i++) { a = simulatePiRho(a); b = simulatePiRho(b); }
    const affected = [];
    let bits = 0;
    for (let i = 0; i < 64; i++) {
      const diff = a[i] ^ b[i];
      if (diff !== 0n) {
        affected.push(i);
        let d = diff; while (d > 0n) { bits += Number(d & 1n); d >>= 1n; }
      }
    }
    results.push({ round: r, positions: affected.length, bits, affected });
  }
  return results;
};

const testDiffusionFromMultiple = (maxStartPos, rounds) => {
  const allAffected = Array(rounds).fill(null).map(() => new Map());
  for (let start = 0; start <= maxStartPos; start++) {
    for (let r = 1; r <= rounds; r++) {
      const s0 = new Array(64).fill(0n); s0[start] = 1n;
      const s1 = new Array(64).fill(0n); s1[start] = 2n;
      let a = s0, b = s1;
      for (let i = 0; i < r; i++) { a = simulatePiRho(a); b = simulatePiRho(b); }
      for (let i = 0; i < 64; i++) {
        if ((a[i] ^ b[i]) !== 0n) {
          const map = allAffected[r - 1];
          map.set(i, (map.get(i) || 0) + 1);
        }
      }
    }
  }
  return allAffected.map((map, idx) => ({
    round: idx + 1,
    affected: Array.from(map.entries()).map(([pos, count]) => ({ pos, count })),
    positions: map.size,
    totalHits: Array.from(map.values()).reduce((a, b) => a + b, 0)
  }));
};

const testFullAvalanche = (rounds) => {
  const avgByRound = [];
  for (let r = 1; r <= rounds; r++) {
    let totalPos = 0, totalBits = 0;
    for (let start = 0; start < 64; start++) {
      const diff = testDiffusionFrom(start, r);
      totalPos += diff[r - 1].positions;
      totalBits += diff[r - 1].bits;
    }
    avgByRound.push({ round: r, avgPositions: (totalPos / 64).toFixed(1), avgBits: (totalBits / 64).toFixed(1), coverage: ((totalPos / 64 / 64) * 100).toFixed(1) });
  }
  return avgByRound;
};

const evaluateQuality = (diffusionMulti, rowMatrix, colMatrix, cycles, maxRounds) => {
  const scores = [];
  let totalScore = 0;

  const fullCoverageRound = diffusionMulti.findIndex(d => d.positions === 64) + 1;
  const coverageScore = fullCoverageRound === 0 ? 0 : Math.max(0, 100 - (fullCoverageRound - 1) * 20);
  scores.push({ name: 'Full Coverage Speed', value: fullCoverageRound > 0 ? `R${fullCoverageRound}` : `Never`, score: coverageScore, max: 100, good: fullCoverageRound <= 2, ideal: 'R1 (64/64 al primo round)', desc: 'Quanti round servono per colpire tutte le 64 posizioni. Ideale: raggiungere 64/64 al round 1.' });
  totalScore += coverageScore;

  const r1Coverage = diffusionMulti[0]?.positions || 0;
  const r1Score = Math.round((r1Coverage / 64) * 100);
  scores.push({ name: 'R1 Coverage', value: `${r1Coverage}/64`, score: r1Score, max: 100, good: r1Coverage >= 48, ideal: '64/64 (100%)', desc: 'Posizioni colpite dopo 1 solo round. Ideale: tutte le 64 posizioni (100%).' });
  totalScore += r1Score;

  const hitsPerPos = diffusionMulti[diffusionMulti.length - 1]?.affected.map(a => a.count) || [];
  const avgHits = hitsPerPos.length > 0 ? hitsPerPos.reduce((a, b) => a + b, 0) / hitsPerPos.length : 0;
  const variance = hitsPerPos.length > 0 ? hitsPerPos.reduce((s, h) => s + Math.pow(h - avgHits, 2), 0) / hitsPerPos.length : 0;
  const stdDev = Math.sqrt(variance);
  const uniformityScore = Math.max(0, Math.round(100 - stdDev * 2));
  scores.push({ name: 'Hit Uniformity', value: `σ=${stdDev.toFixed(1)}`, score: uniformityScore, max: 100, good: stdDev < 10, ideal: 'σ=0 (tutti uguali)', desc: 'Deviazione standard degli hit per posizione. Ideale: σ=0, tutti i punti colpiti lo stesso numero di volte.' });
  totalScore += uniformityScore;

  const rowZeros = rowMatrix.flat().filter(v => v === 0).length;
  const rowScore = Math.round((1 - rowZeros / 64) * 100);
  scores.push({ name: 'Row Connectivity', value: `${64 - rowZeros}/64`, score: rowScore, max: 100, good: rowZeros === 0, ideal: '64/64 (nessuno 0)', desc: 'Celle non-zero nella matrice Row Trans. Ideale: tutte 64 celle con valore ≥1, nessuno 0.' });
  totalScore += rowScore;

  const colZeros = colMatrix.flat().filter(v => v === 0).length;
  const colScore = Math.round((1 - colZeros / 64) * 100);
  scores.push({ name: 'Col Connectivity', value: `${64 - colZeros}/64`, score: colScore, max: 100, good: colZeros === 0, ideal: '64/64 (nessuno 0)', desc: 'Celle non-zero nella matrice Col Trans. Ideale: tutte 64 celle con valore ≥1, nessuno 0.' });
  totalScore += colScore;

  const fixedPoints = cycles.filter(c => c.length === 1).length;
  const fixedScore = fixedPoints === 0 ? 100 : Math.max(0, 100 - fixedPoints * 25);
  scores.push({ name: 'No Fixed Points', value: fixedPoints === 0 ? '✓' : `${fixedPoints}`, score: fixedScore, max: 100, good: fixedPoints === 0, ideal: '0 (nessun punto fisso)', desc: 'Posizioni dove PI3D[i]=i. Ideale: zero punti fissi, ogni posizione deve spostarsi.' });
  totalScore += fixedScore;

  const longestCycle = Math.max(...cycles.map(c => c.length));
  const cycleScore = Math.round((longestCycle / 64) * 100);
  scores.push({ name: 'Cycle Length', value: `${longestCycle}/64`, score: cycleScore, max: 100, good: longestCycle >= 32, ideal: '64 (un solo ciclo)', desc: 'Lunghezza del ciclo più lungo. Ideale: 64 (tutta la permutazione è un unico ciclo).' });
  totalScore += cycleScore;

  const rotVals = ROT3D.flat(2);
  const rotZeros = rotVals.filter(v => v === 0).length;
  const rotScore = rotZeros === 0 ? 100 : Math.max(0, 100 - rotZeros * 20);
  scores.push({ name: 'ROT3D No Zeros', value: rotZeros === 0 ? '✓' : `${rotZeros}`, score: rotScore, max: 100, good: rotZeros === 0, ideal: '0 zeri (tutte rotazioni >0)', desc: 'Rotazioni con valore 0 non mescolano i bit. Ideale: nessuna rotazione a 0.' });
  totalScore += rotScore;

  const maxTotal = scores.length * 100;
  const grade = totalScore >= maxTotal * 0.9 ? 'A' : totalScore >= maxTotal * 0.8 ? 'B' : totalScore >= maxTotal * 0.7 ? 'C' : totalScore >= maxTotal * 0.6 ? 'D' : 'F';
  return { scores, totalScore, maxTotal, grade };
};

    <div className="inline-block mr-4 mb-2">
      <div className="text-[10px] font-bold mb-1">→ {positionsEverHit}/64 hit</div>
      {[0, 1, 2, 3, 4, 5, 6, 7].map(y => (

          })}

};

            {avalanche.map(a => (
              <div key={a.round} className={+a.coverage > 90 ? 'text-green-600 font-bold' : +a.coverage > 50 ? 'text-yellow-600' : 'text-red-600'}>
                R{a.round}: {a.avgPositions}/64 ({a.coverage}%)
              </div>
            ))}
          </div>
        </div>
      </div>

      <div className="flex flex-wrap mb-4">
        <Grid8x8 title="PI3D[i]" getData={i => PI3D[i]} getColor={(i, v) => Math.floor(v / 8) === Math.floor(i / 8) ? 'bg-red-200' : 'bg-blue-100'} infoType="pi" />
        <Grid8x8 title="PI3D_INV[i]" getData={i => PI3D_INV[i]} getColor={() => 'bg-purple-100'} />
        <Grid8x8 title="ROT3D" getData={i => getRotValue(i)} getColor={(i, v) => v === 0 ? 'bg-red-300' : v < 32 ? 'bg-green-100' : 'bg-blue-100'} infoType="rot" />
        <Matrix8x8 data={rowMatrix} title={`Row Trans (${maxRounds}r)`} infoType="row" />
        <Matrix8x8 data={colMatrix} title={`Col Trans (${maxRounds}r)`} infoType="col" />
      </div>

      <Planes4x4 title="ROT3D 4 Planes" />

      <div className="bg-gray-50 p-2 rounded mb-4">
        <div className="font-bold text-xs mb-2">
          Diffusion Test | Position:
          <input type="number" min={0} max={63} value={startPos} onChange={e => setStartPos(+e.target.value)} className="w-12 mx-2 border rounded px-1" />
          | Rounds:
          <input type="number" min={1} max={20} value={maxRounds} onChange={e => setMaxRounds(+e.target.value)} className="w-12 mx-2 border rounded px-1" />
        </div>

        <div className="flex flex-wrap mb-3">
          <div className="mr-6">
            <div className="text-[10px] font-bold mb-1 text-blue-700">Cumulative Single (pos {startPos})</div>
            <CumulativeSingleVis diffusionResults={diffusionSingle} maxRounds={maxRounds} />
          </div>
          <div className="mr-6">
            <div className="text-[10px] font-bold mb-1 text-green-700">Cumulative Multi (0-{startPos})</div>
            <CumulativeMultiVis diffusionResults={diffusionMulti} maxStartPos={startPos} />
          </div>
        </div>

        <div className="grid grid-cols-2 gap-4">
          <div>
            <div className="text-[10px] font-bold mb-1 text-blue-700">Per-Round Single:</div>
            <div className="flex flex-wrap">
              {diffusionSingle.map(d => <StateVisSingle key={d.round} affected={d.affected} round={d.round} />)}
            </div>
            <div className="text-[10px] mt-1">
              {diffusionSingle.map(d => (
                <span key={d.round} className={`mr-2 ${d.positions >= 60 ? 'text-green-600 font-bold' : d.positions >= 32 ? 'text-yellow-600' : 'text-red-600'}`}>R{d.round}:{d.positions}</span>
              ))}
            </div>
          </div>
          <div>
            <div className="text-[10px] font-bold mb-1 text-green-700">Per-Round Multi:</div>
            <div className="flex flex-wrap">
              {diffusionMulti.map(d => <StateVisMulti key={d.round} affected={d.affected} round={d.round} maxStartPos={startPos} />)}
            </div>
            <div className="text-[10px] mt-1">
              {diffusionMulti.map(d => (
                <span key={d.round} className={`mr-2 ${d.positions >= 60 ? 'text-green-600 font-bold' : d.positions >= 32 ? 'text-yellow-600' : 'text-red-600'}`}>R{d.round}:{d.positions}</span>
              ))}
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}
